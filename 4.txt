Q3. Input the date of issue and date of return for a book. Calculate and display the fine with the appropriate message using a PL/SQL block. The fine is charged as per Table 8.1.
The manual's fine table specifies:
- 7 days: NIL
- 8–15 days: Rs. 1/day
- 16–30 days: Rs. 2/day
- After 30 days: Rs. 5.00

DECLARE
    v_issue_date DATE := DATE '2026-01-01';
    v_return_date DATE := DATE '2026-01-20';
    v_days NUMBER;
    v_fine NUMBER := 0;
BEGIN
    v_days := v_return_date - v_issue_date;

    IF v_days <= 7 THEN
        v_fine := 0;
    ELSIF v_days <= 15 THEN
        v_fine := (v_days - 7) * 1;
    ELSIF v_days <= 30 THEN
        v_fine := (8 * 1) + ((v_days - 15) * 2);
    ELSE
        v_fine := (8 * 1) + (15 * 2)
                   + ((v_days - 30) * 5);
    END IF;

    DBMS_OUTPUT.PUT_LINE('Late days: ' || v_days);
    DBMS_OUTPUT.PUT_LINE('Fine: Rs. ' || v_fine);
END;
/

Q6. Write a PL/SQL block to find the student with max. GPA without using aggregate function.

DECLARE
    v_max_gpa NUMBER;
    v_max_roll NUMBER;
    v_gpa NUMBER;
BEGIN
    SELECT RollNo, GPA
    INTO v_max_roll, v_max_gpa
    FROM StudentTable
    WHERE RollNo = 1;

    FOR r IN 2..5 LOOP
        SELECT GPA INTO v_gpa
        FROM StudentTable
        WHERE RollNo = r;

        IF v_gpa > v_max_gpa THEN
            v_max_gpa := v_gpa;
            v_max_roll := r;
        END IF;
    END LOOP;

    DBMS_OUTPUT.PUT_LINE('Roll No: ' || v_max_roll);
    DBMS_OUTPUT.PUT_LINE('Maximum GPA: ' || v_max_gpa);
END;

        SELECT ID, salary

Q11. Print the Course details and the total number of students registered for each course along with the course details — (Course-id, title, dept-name, credits, instructor_name, building, room-number, time-slot-id, tot_student_no).

BEGIN
    FOR r IN (
        SELECT C.course_id,
               C.title,
               C.dept_name,
               C.credits,
               I.name AS instructor_name,
               S.building,
               S.room_number,
               S.time_slot_id,
               COUNT(T.studID) AS tot_student_no
        FROM Course C
        LEFT JOIN Section S
            ON C.course_id = S.course_id
        LEFT JOIN Teaches TE
            ON S.course_id = TE.course_id
           AND S.section_id = TE.section_id
           AND S.semester = TE.semester
           AND S.year = TE.year
        LEFT JOIN Instructor I
            ON TE.ID = I.ID
        LEFT JOIN Takes T
            ON S.course_id = T.course_id
           AND S.section_id = T.section_id
           AND S.semester = T.semester
           AND S.year = T.year
        GROUP BY C.course_id, C.title, C.dept_name, C.credits,
                 I.name, S.building, S.room_number, S.time_slot_id
    ) LOOP
        DBMS_OUTPUT.PUT_LINE(
            'Course: ' || r.course_id ||
            ', Title: ' || r.title ||
            ', Department: ' || r.dept_name ||
            ', Credits: ' || r.credits ||
            ', Instructor: ' || NVL(r.instructor_name, 'Not assigned') ||
            ', Building: ' || NVL(r.building, 'Not assigned') ||
            ', Room: ' || NVL(TO_CHAR(r.room_number), 'N/A') ||
            ', Time Slot: ' || NVL(r.time_slot_id, 'N/A') ||
            ', Students: ' || r.tot_student_no
        );
    END LOOP;
END;

Q1. Based on the University database Schema in Lab 2, write a row trigger that records along with the time any change made in the Takes (ID, course-id, sec-id, semester, year, grade) table in log_change_Takes (Time_Of_Change, ID, courseid, sec-id, semester, year, grade).
/The manual names the column ID in the log table, while the Takes table uses studID. The code below records the student ID in that log column.

Create table
CREATE TABLE log_change_Takes (
    Time_Of_Change TIMESTAMP,
    ID NUMBER,
    courseid VARCHAR2(10),
    sec_id NUMBER,
    semester VARCHAR2(10),
    year NUMBER,
    grade VARCHAR2(2)
);


/Create trigger

CREATE OR REPLACE TRIGGER trg_takes_log
AFTER INSERT OR UPDATE OR DELETE ON Takes
FOR EACH ROW
BEGIN
    IF INSERTING OR UPDATING THEN
        INSERT INTO log_change_Takes
        VALUES (
            SYSTIMESTAMP,
            :NEW.studID,
            :NEW.course_id,
            :NEW.section_id,
            :NEW.semester,
            :NEW.year,
            :NEW.grade
        );
    ELSIF DELETING THEN
        INSERT INTO log_change_Takes
        VALUES (
            SYSTIMESTAMP,
            :OLD.studID,
            :OLD.course_id,
            :OLD.section_id,
            :OLD.semester,
            :OLD.year,
            :OLD.grade
        );
    END IF;
END;
/

Q3. Based on the University Schema, write a database trigger on Instructor that checks the following:
- The name of the instructor is a valid name containing only alphabets.
- The salary of an instructor is not zero and is positive.
- The salary does not exceed the budget of the department to which the instructor belongs.

CREATE OR REPLACE TRIGGER trg_validate_instructor
BEFORE INSERT OR UPDATE ON Instructor
FOR EACH ROW
DECLARE
    v_budget Department.budget%TYPE;
BEGIN
    IF NOT REGEXP_LIKE(:NEW.name, '^[A-Za-z]+$') THEN
        RAISE_APPLICATION_ERROR(
            -20001,
            'Instructor name must contain only alphabets'
        );
    END IF;

    IF :NEW.salary <= 0 THEN
        RAISE_APPLICATION_ERROR(
            -20002,
            'Salary must be positive'
        );
    END IF;

    SELECT budget INTO v_budget
    FROM Department
    WHERE dept_name = :NEW.dept_name;

    IF :NEW.salary > v_budget THEN
        RAISE_APPLICATION_ERROR(
            -20003,
            'Salary cannot exceed department budget'
        );
    END IF;
END;
/

Q4. Create a transparent audit system for a table Client_master (client_no, name, address, Bal_due). The system must keep track of the records that are being deleted or updated. When a record is deleted or modified, the original record details and the date of operation are stored in the auditclient (client_no, name, bal_due, operation, userid, opdate) table, then the delete or update is allowed to go through.

Create tables
CREATE TABLE Client_master (
    client_no NUMBER PRIMARY KEY,
    name VARCHAR2(30),
    address VARCHAR2(100),
    Bal_due NUMBER(10,2)
);

CREATE TABLE auditclient (
    client_no NUMBER,
    name VARCHAR2(30),
    bal_due NUMBER(10,2),
    operation VARCHAR2(10),
    userid VARCHAR2(30),
    opdate DATE
);


INSERT INTO Client_master

INSERT INTO Client_master

INSERT INTO Client_master

/Create audit trigger

CREATE OR REPLACE TRIGGER trg_client_audit
BEFORE UPDATE OR DELETE ON Client_master
FOR EACH ROW
BEGIN
    IF UPDATING THEN
        INSERT INTO auditclient
        VALUES (
            :OLD.client_no,
            :OLD.name,
            :OLD.Bal_due,
            'UPDATE',
            USER,
            SYSDATE
        );
    ELSIF DELETING THEN
        INSERT INTO auditclient
        VALUES (
            :OLD.client_no,
            :OLD.name,
            :OLD.Bal_due,
            'DELETE',

            SYSDATE
